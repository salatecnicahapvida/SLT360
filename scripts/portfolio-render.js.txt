function renderPortfolio() {
  const allWorks = budgetWorks();

  const yearOf = (work) => {
    const versions = Array.isArray(work?.ev?.versions) ? work.ev.versions : [];
    const rawDate = versions[0]?.data || work?.data || work?.createdAt || "";
    const match = String(rawDate).match(/(20\d{2})/);
    return match ? match[1] : "Não informado";
  };
  const categoryOf = (work) => work?.classificacaoObra || work?.tipologiaObra || work?.tipoUnidade || "Não informada";
  const typeOf = (work) => work?.tipoUnidade || "Não informado";
  const typologyOf = (work) => work?.tipologiaObra || "Não informada";
  const regionOf = (work) => work?.regiao || "Não informada";
  const ufOf = (work) => work?.uf || "Não informada";
  const statusOf = (work) => work?.ev?.status || "Sem EV";
  const originOf = (work) => work?._historicalBudgetWork ? "Histórico" : "Atual";
  const areaOf = (work) => Number(work?.areaEquivalente || work?.areaConstruida || 0);
  const totalOf = (work) => {
    if (work?._historicalBudgetWork) {
      const historicalTotal = Number(work?.ev?.versions?.[0]?.valorTotal || 0);
      if (historicalTotal) return historicalTotal;
    }
    const totals = workTotals(work);
    return Number(totals.orcado || 0) + Number(totals.aditivado || 0);
  };
  const optionValues = (mapper) => [...new Set(allWorks.map(mapper).filter(Boolean))].sort((a, b) => String(a).localeCompare(String(b), "pt-BR", { numeric: true }));
  const selected = (key, value) => String(portfolioQuickFilters[key] || "") === String(value) ? "selected" : "";

  const terms = normalizeSearchText([searchTerm, portfolioQuickFilters.query].filter(Boolean).join(" ")).trim().split(/\s+/).filter(Boolean);
  const rows = allWorks.filter((work) => {
    if (terms.length && !terms.every((term) => workSearchText(work).includes(term))) return false;
    if (portfolioQuickFilters.year && yearOf(work) !== portfolioQuickFilters.year) return false;
    if (portfolioQuickFilters.category && categoryOf(work) !== portfolioQuickFilters.category) return false;
    if (portfolioQuickFilters.tipoUnidade && typeOf(work) !== portfolioQuickFilters.tipoUnidade) return false;
    if (portfolioQuickFilters.tipologia && typologyOf(work) !== portfolioQuickFilters.tipologia) return false;
    if (portfolioQuickFilters.regional && regionOf(work) !== portfolioQuickFilters.regional) return false;
    if (portfolioQuickFilters.uf && ufOf(work) !== portfolioQuickFilters.uf) return false;
    if (portfolioQuickFilters.evStatus && statusOf(work) !== portfolioQuickFilters.evStatus) return false;
    if (portfolioQuickFilters.origin && originOf(work) !== portfolioQuickFilters.origin) return false;
    return true;
  });

  const totalOrcado = rows.reduce((sum, work) => sum + totalOf(work), 0);
  const withArea = rows.filter((work) => areaOf(work) > 0);
  const areaTotal = withArea.reduce((sum, work) => sum + areaOf(work), 0);
  const totalWithArea = withArea.reduce((sum, work) => sum + totalOf(work), 0);
  const custoM2 = areaTotal > 0 ? totalWithArea / areaTotal : 0;

  return `
    ${renderWorksToolbar("portfolio", "Portfólio de Obras", "Visão consolidada da carteira de obras com filtros aplicados aos indicadores.", "")}

    <section class="panel portfolio-filter-panel">
      <div class="panel-heading">
        <div>
          <h2>Filtros</h2>
          <p class="panel-subtitle">Refine os indicadores por ano, categoria, tipo e demais informações disponíveis.</p>
        </div>
        <button class="secondary-action" type="button" data-action="clear-portfolio-filters">Limpar filtros</button>
      </div>
      <div class="portfolio-filter-grid">
        <label class="portfolio-filter-field portfolio-filter-field--search">
          <span>Buscar</span>
          <input data-portfolio-search value="${escapeAttribute(portfolioQuickFilters.query || "")}" placeholder="Obra, código, cidade, região..." />
        </label>
        <label class="portfolio-filter-field">
          <span>Ano</span>
          <select data-portfolio-quick-filter="year">
            <option value="">Todos</option>
            ${optionValues(yearOf).map((value) => `<option value="${escapeAttribute(value)}" ${selected("year", value)}>${escapeAttribute(value)}</option>`).join("")}
          </select>
        </label>
        <label class="portfolio-filter-field">
          <span>Categoria</span>
          <select data-portfolio-quick-filter="category">
            <option value="">Todas</option>
            ${optionValues(categoryOf).map((value) => `<option value="${escapeAttribute(value)}" ${selected("category", value)}>${escapeAttribute(value)}</option>`).join("")}
          </select>
        </label>
        <label class="portfolio-filter-field">
          <span>Tipo</span>
          <select data-portfolio-quick-filter="tipoUnidade">
            <option value="">Todos</option>
            ${optionValues(typeOf).map((value) => `<option value="${escapeAttribute(value)}" ${selected("tipoUnidade", value)}>${escapeAttribute(value)}</option>`).join("")}
          </select>
        </label>
        <label class="portfolio-filter-field">
          <span>Tipologia</span>
          <select data-portfolio-quick-filter="tipologia">
            <option value="">Todas</option>
            ${optionValues(typologyOf).map((value) => `<option value="${escapeAttribute(value)}" ${selected("tipologia", value)}>${escapeAttribute(value)}</option>`).join("")}
          </select>
        </label>
        <label class="portfolio-filter-field">
          <span>Região</span>
          <select data-portfolio-quick-filter="regional">
            <option value="">Todas</option>
            ${optionValues(regionOf).map((value) => `<option value="${escapeAttribute(value)}" ${selected("regional", value)}>${escapeAttribute(value)}</option>`).join("")}
          </select>
        </label>
        <label class="portfolio-filter-field">
          <span>UF</span>
          <select data-portfolio-quick-filter="uf">
            <option value="">Todas</option>
            ${optionValues(ufOf).map((value) => `<option value="${escapeAttribute(value)}" ${selected("uf", value)}>${escapeAttribute(value)}</option>`).join("")}
          </select>
        </label>
        <label class="portfolio-filter-field">
          <span>Status do EV</span>
          <select data-portfolio-quick-filter="evStatus">
            <option value="">Todos</option>
            ${optionValues(statusOf).map((value) => `<option value="${escapeAttribute(value)}" ${selected("evStatus", value)}>${escapeAttribute(value)}</option>`).join("")}
          </select>
        </label>
        <label class="portfolio-filter-field">
          <span>Origem</span>
          <select data-portfolio-quick-filter="origin">
            <option value="">Todas</option>
            ${optionValues(originOf).map((value) => `<option value="${escapeAttribute(value)}" ${selected("origin", value)}>${escapeAttribute(value)}</option>`).join("")}
          </select>
        </label>
      </div>
    </section>

    <section class="kpi-grid portfolio-kpis portfolio-kpis--summary">
      ${kpi("Obras no portfólio", String(rows.length), `${allWorks.length} obra(s) na base total`, "blue")}
      ${kpi("Total orçado", money(totalOrcado), "Valores das obras no filtro atual", "blue")}
      ${kpi("Custo por m²", custoM2 ? `${money(custoM2)}/m²` : "—", `${withArea.length} obra(s) com área informada`, "green")}
    </section>
  `;
}
