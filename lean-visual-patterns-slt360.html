// Chart.js: barra horizontal com plugin simples para meta
const targetMarkerPlugin = {
  id: "targetMarker",
  afterDatasetsDraw(chart, args, opts) {
    const scale = chart.scales.x;
    const x = scale.getPixelForValue(opts.target);
    const {top, bottom} = chart.chartArea;
    const ctx = chart.ctx;
    ctx.save(); ctx.strokeStyle = "#08213f"; ctx.lineWidth = 2;
    ctx.beginPath(); ctx.moveTo(x, top); ctx.lineTo(x, bottom); ctx.stroke(); ctx.restore();
  }
};

new Chart(ctx, {
  type: "bar",
  data: {labels: ["Skarth"], datasets: [{data: [92], backgroundColor: ["#FF212C"]}]},
  options: {indexAxis: "y", scales: {x: {min: 0, max: 100}}},
  plugins: [targetMarkerPlugin],
});
